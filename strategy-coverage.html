// CreateStrategy creates a new strategy
func (s *strategyService) CreateStrategy(ctx context.Context, req *dto.CreateStrategyRequest, userID uuid.UUID) (*dto.StrategyResponse, error) {
        // Validate request
        if err := req.Validate(); err != nil {
                return nil, ErrInvalidRequest
        }

        // Validate entry expression
        if err := validateExpression(req.EntryExpression); err != nil {
                return nil, ErrInvalidExpression
        }

        // Validate exit expression if provided
        if req.ExitExpression != "" {
                if err := validateExpression(req.ExitExpression); err != nil {
                        return nil, ErrInvalidExpression
                }
        }

        // Convert DTO to domain model
        strategy := req.ToDomain(userID)

        // Save strategy
        err := s.strategyRepo.Create(ctx, strategy)
        if err != nil {
                return nil, ErrStrategyCreateFailed
        }

        // Convert domain model to response DTO
        return dto.FromDomain(strategy), nil
}

// GetStrategyByID retrieves a strategy by its ID
func (s *strategyService) GetStrategyByID(ctx context.Context, id uuid.UUID, userID uuid.UUID) (*dto.StrategyResponse, error) {
        strategy, err := s.strategyRepo.GetByID(ctx, id)
        if err != nil {
                return nil, ErrStrategyNotFound
        }

        // Check if user has access to the strategy
        if strategy.UserID != userID {
                return nil, ErrUnauthorized
        }

        return dto.FromDomain(strategy), nil
}

// GetUserStrategies retrieves all strategies for a user
func (s *strategyService) GetUserStrategies(ctx context.Context, userID uuid.UUID) ([]*dto.StrategyResponse, error) {
        strategies, err := s.strategyRepo.GetByUserID(ctx, userID)
        if err != nil {
                return nil, err
        }

        // Convert domain models to response DTOs
        var strategyResponses []*dto.StrategyResponse
        for _, strategy := range strategies {
                strategyResponses = append(strategyResponses, dto.FromDomain(strategy))
        }

        return strategyResponses, nil
}

// UpdateStrategy updates an existing strategy
func (s *strategyService) UpdateStrategy(ctx context.Context, req *dto.UpdateStrategyRequest, userID uuid.UUID) (*dto.StrategyResponse, error) {
        // Validate request
        if err := req.Validate(); err != nil {
                return nil, ErrInvalidRequest
        }

        // Retrieve the strategy
        existingStrategy, err := s.strategyRepo.GetByID(ctx, req.ID)
        if err != nil {
                return nil, ErrStrategyNotFound
        }

        // Check if user has access to the strategy
        if existingStrategy.UserID != userID {
                return nil, ErrUnauthorized
        }

        // Validate entry expression
        if err := validateExpression(req.EntryExpression); err != nil {
                return nil, ErrInvalidExpression
        }

        // Validate exit expression if provided
        if req.ExitExpression != "" {
                if err := validateExpression(req.ExitExpression); err != nil {
                        return nil, ErrInvalidExpression
                }
        }

        // Convert request DTO to domain model
        updatedStrategy := req.ToDomain()

        // Preserve the original user ID and creation timestamp
        updatedStrategy.UserID = existingStrategy.UserID
        updatedStrategy.CreatedAt = existingStrategy.CreatedAt
        updatedStrategy.UpdatedAt = time.Now()
        updatedStrategy.IsActive = existingStrategy.IsActive

        // Save updated strategy
        err = s.strategyRepo.Update(ctx, updatedStrategy)
        if err != nil {
                return nil, ErrStrategyUpdateFailed
        }

        return dto.FromDomain(updatedStrategy), nil
}

                filters["created_after"] = *filter.CreatedAfter
        }

        if filter.CreatedBefore != nil {
                filters["created_before"] = *filter.CreatedBefore
        }

        // Search strategies

        for _, strategy := range strategies {

        )

// SearchStrategies searches strategies with filter criteria
func (r *StrategyRepository) SearchStrategies(ctx context.Context, filters map[string]interface{}) ([]*model.Strategy, error) {
        // Start with base query
        query := `
                SELECT 
                        id, user_id, name, description, position_type, instruments, 
                        chart_type, chart_timeframe, trade_type, qty, entry_expression, 
                        exit_expression, target_profit, target_profit_type, stop_loss, 
                        stop_loss_type, trailing_stop_loss, trailing_stop_loss_type, 
                        exit_type, capital, max_trades, entry_start_time, entry_end_time, 
                        start_date, end_date, created_at, updated_at, is_active
                FROM strategy
                WHERE 1=1
        `

        // Dynamically build the where clause based on filters
        args := []interface{}{}
        argPosition := 1
        var whereClauses []string

        // Process filters
        for key, value := range filters {
                switch key {
                case "user_id":
                        whereClauses = append(whereClauses, fmt.Sprintf("user_id = $%d", argPosition))
                        args = append(args, value)
                        argPosition++
                case "name":
                        whereClauses = append(whereClauses, fmt.Sprintf("name ILIKE $%d", argPosition))
                        args = append(args, "%"+value.(string)+"%")
                        argPosition++
                case "position_type":
                        whereClauses = append(whereClauses, fmt.Sprintf("position_type = $%d", argPosition))
                        args = append(args, value)
                        argPosition++
                case "instrument":
                        whereClauses = append(whereClauses, fmt.Sprintf("$%d = ANY(instruments)", argPosition))
                        args = append(args, value)
                        argPosition++
                case "chart_type":
                        whereClauses = append(whereClauses, fmt.Sprintf("chart_type = $%d", argPosition))
                        args = append(args, value)
                        argPosition++
                case "chart_timeframe":
                        whereClauses = append(whereClauses, fmt.Sprintf("chart_timeframe = $%d", argPosition))
                        args = append(args, value)
                        argPosition++
                case "is_active":
                        whereClauses = append(whereClauses, fmt.Sprintf("is_active = $%d", argPosition))
                        args = append(args, value)
                        argPosition++
                case "created_after":
                        whereClauses = append(whereClauses, fmt.Sprintf("created_at >= $%d", argPosition))
                        args = append(args, value)
                        argPosition++
                case "created_before":
                        whereClauses = append(whereClauses, fmt.Sprintf("created_at <= $%d", argPosition))
                        args = append(args, value)
                        argPosition++
                }
        }

        // Append the where clauses to the query if any
        if len(whereClauses) > 0 {
                query += " AND " + strings.Join(whereClauses, " AND ")
        }

        // Add ordering
        query += " ORDER BY created_at DESC"

        // Execute the query
        rows, err := r.db.QueryContext(ctx, query, args...)
        if err != nil {
                return nil, fmt.Errorf("%w: %v", ErrDatabaseError, err)
        }
        defer rows.Close()

        var strategies []*model.Strategy

        for rows.Next() {
                var strategy model.Strategy
                var instruments pq.Int32Array

                err := rows.Scan(
                        &strategy.ID,
                        &strategy.UserID,
                        &strategy.Name,
                        &strategy.Description,
                        &strategy.PositionType,
                        pq.Array(&instruments),
                        &strategy.ChartType,
                        &strategy.ChartTimeframe,
                        &strategy.TradeType,
                        &strategy.Qty,
                        &strategy.EntryExpression,
                        &strategy.ExitExpression,
                        &strategy.TargetProfit,
                        &strategy.TargetProfitType,
                        &strategy.StopLoss,
                        &strategy.StopLossType,
                        &strategy.TrailingStopLoss,
                        &strategy.TrailingStopLossType,
                        &strategy.ExitType,
                        &strategy.Capital,
                        &strategy.MaxTrades,
                        &strategy.EntryStartTime,
                        &strategy.EntryEndTime,
                        &strategy.StartDate,
                        &strategy.EndDate,
                        &strategy.CreatedAt,
                        &strategy.UpdatedAt,
                        &strategy.IsActive,
                )

                if err != nil {
                        return nil, fmt.Errorf("%w: %v", ErrDatabaseError, err)
                }

                strategy.Instruments = instruments
                strategies = append(strategies, &strategy)
        }

        if err = rows.Err(); err != nil {
                return nil, fmt.Errorf("%w: %v", ErrDatabaseError, err)
        }

        return strategies, nil
}
